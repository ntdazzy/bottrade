#ifndef BOT3_OPEN_STATUS_PANEL_MQH
#define BOT3_OPEN_STATUS_PANEL_MQH
#include "Bot3NewsSelector.mqh"

#ifndef BOT3_PANEL_TEXT_CASTS_MQH
#define BOT3_PANEL_TEXT_CASTS_MQH
string Bot3PanelDoubleText(const double value) { return (string)value; }
string Bot3PanelIntegerText(const int value) { return (string)value; }
string Bot3PanelTimeText(const datetime value) { return (string)value; }
#endif

void Bot3UpdateOpenStatusPanelFields(const uchar &show,const uchar &show_tester,const int &seed,const string &symbol,
                              const double &balance,const int &frequency,const int &mode,const double &manual_lots,
                              const double &allowed_dd,const double &strategy_risk,const Bot3NewsEvents &events,
                              const uchar &nfp_enabled,const int &nfp_after,const uchar &interest_enabled,
                              const int &interest_after,const uchar &cpi_enabled,const int &cpi_after,
                              Bot3NewsSelection &selection,ulong &last_ticket)
{
   if(show==0) return;
   const long tester=MQLInfoInteger((ENUM_MQL_INFO_INTEGER)6);
   if(tester==1 && show_tester==0) return;
   double open_pl=0.0;
   const int total=PositionsTotal();
   if(total>0)
   {
      long accepted[15];
      for(int i=0;i<15;i++) accepted[i]=(long)(int)((uint)seed+(uint)(i+1));
      int index=0;
      do
      {
         last_ticket=PositionGetTicket(index);
         if(last_ticket!=0 && PositionGetString((ENUM_POSITION_PROPERTY_STRING)0)==symbol)
         {
            bool magic_match=false;
            for(int i=0;i<15;i++)
            {
               if(PositionGetInteger((ENUM_POSITION_PROPERTY_INTEGER)12)==accepted[i])
               { magic_match=true;break; }
            }
            if(magic_match && (PositionGetInteger((ENUM_POSITION_PROPERTY_INTEGER)2)==0 ||
                               PositionGetInteger((ENUM_POSITION_PROPERTY_INTEGER)2)==1))
            {
               PositionSelectByTicket(last_ticket);
               const double first=PositionGetDouble((ENUM_POSITION_PROPERTY_DOUBLE)10);
               const double second=PositionGetDouble((ENUM_POSITION_PROPERTY_DOUBLE)9);
               SetUserError(4014);
               const double part=(first+second)+0.0;
               open_pl+=part;
            }
         }
         index++;
      } while(index<PositionsTotal());
   }
   const string value=DoubleToString(open_pl,2);
   string index_text=IntegerToString(0,0,32);
   ObjectSetString(0,"lineopl"+index_text,(ENUM_OBJECT_PROPERTY_STRING)6,"Open P/L: "+value);
   const string balance_text=DoubleToString(balance,2);
   index_text=IntegerToString(0,0,32);
   ObjectSetString(0,"linea"+index_text,(ENUM_OBJECT_PROPERTY_STRING)6,"Account Balance: "+balance_text);
   Bot3UpdateNewsSelectionFields(nfp_enabled,events.nfp,nfp_after,interest_enabled,events.interest,interest_after,
                                 cpi_enabled,events.cpi,cpi_after,selection);
   string news_text;
   if(selection.block_time!=0)
      news_text="Next News: "+selection.label+", "+Bot3PanelTimeText((datetime)selection.block_time);
   else news_text="no news coming up";
   index_text=IntegerToString(0,0,32);
   ObjectSetString(0,"linenfp"+index_text,(ENUM_OBJECT_PROPERTY_STRING)6,news_text);
   string frequency_text;
   if(frequency==0) frequency_text="extreme conservative";
   else if(frequency==1) frequency_text="conservative";
   else if(frequency==2) frequency_text="moderate";
   else if(frequency==3) frequency_text="intense";
   else if(frequency==4) frequency_text="extreme";
   else frequency_text="manual strategy selection";
   ObjectSetString(0,"lines",(ENUM_OBJECT_PROPERTY_STRING)6,"Trade Frequency: "+frequency_text);
   string risk_text;
   if(mode==3) risk_text="Max risk per strategy: "+Bot3PanelDoubleText(strategy_risk)+"%";
   else if(mode==1234) risk_text="Max allowed DD: "+Bot3PanelDoubleText(allowed_dd)+"%";
   else risk_text="Manual lotsize: "+Bot3PanelDoubleText(manual_lots)+"lots";
   ObjectSetString(0,"linet",(ENUM_OBJECT_PROPERTY_STRING)6,risk_text);
}

void Bot3UpdateOpenStatusPanel(const uchar &show,const uchar &show_tester,const int &seed,const string &symbol,
                              const double &balance,const int &frequency,const int &mode,const double &manual_lots,
                              const double &allowed_dd,const double &strategy_risk,const Bot3NewsEvents &events,
                              const Bot3NewsWindows &windows,Bot3NewsSelection &selection,ulong &last_ticket)
{
   Bot3UpdateOpenStatusPanelFields(show,show_tester,seed,symbol,balance,frequency,mode,manual_lots,allowed_dd,
                                   strategy_risk,events,windows.nfp_enabled,windows.nfp_after,windows.interest_enabled,
                                   windows.interest_after,windows.cpi_enabled,windows.cpi_after,selection,last_ticket);
}
