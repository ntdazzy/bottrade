def encode(rows,bot):
    raw=bytearray(struct.pack('<3i',MAGIC,1,len(rows)))
    for c in rows:
        raw+=flow.encode([c],bot)[12:]
        for n,k,_,_ in fields(bot):raw+=struct.pack('<Q',base.bits(c['active'][n]) if k=='d' else c['active'][n]&MASK)
    return bytes(raw)
