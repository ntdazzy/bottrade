#ifndef BOT3_INITIALIZATION_GATE_MQH
#define BOT3_INITIALIZATION_GATE_MQH

struct Bot3InitializationGateState
{
   uchar active;
   uchar dispatch_mode;
   uchar m1_enabled;
};

bool Bot3InitializeGate(const bool authorized,const int tester_mode,const int backtest_speed,
                        Bot3InitializationGateState &state)
{
   state.active=0;
   if(!authorized) return false;
   state.active=1;
   if(tester_mode==1)
   {
      if(backtest_speed==2) { state.dispatch_mode=1;state.m1_enabled=1; }
      else if(backtest_speed==3) { state.dispatch_mode=0;state.m1_enabled=1; }
   }
   return true;
}

#endif
