// Dispatcher context before preparation. Contract: strategy index 0..7.
#ifndef BOT1_STRATEGY_CONTEXT_MQH
#define BOT1_STRATEGY_CONTEXT_MQH
#include "Bot1Currency.mqh"
struct Bot1StrategyContextInputs
{
   uchar keep_comment;
   string comment_prefix;
   int base_magic;
   string symbol_suffix;
};
struct Bot1StrategyContextState
{
   string order_comment;
   int magic;
   double converted_value;
};
double Bot1StrategyContextAmount(const double amount,const string suffix)
{
   // The USD check has no conversion, but its account reads still occur.
   if(AccountInfoString(ACCOUNT_CURRENCY)!="USD") AccountInfoString(ACCOUNT_CURRENCY);
   const string codes[]={"EUR","GBP","AUD","JPY","YEN","CHF","CHF","HKD","RUB","SGD",
                         "BTC","ETH","BCH","BCC","XRP","LTC","XMR","DSH","EOS","TRX","ADA","BSV","XLM","GLD","ZEC","XEM"};
   const string lower[]={"eur","gbp","aud","jpy","yen","chf","chf","hkd","rub","sgd",
                         "btc","eth","bch","bcc","xrp","ltc","xmr","dsh","eos","trx","ada","bsv","xlm","gld","zec","xem"};
   const string pairs[]={"EURUSD","GBPUSD","AUDUSD","USDJPY","USDJPY","USDCHF","USDCHF","USDHKD","USDRUB","USDSGD",
                         "BTCUSD","ETHUSD","BCHUSD","BCCUSD","XRPUSD","LTCUSD","XMRUSD","DSHUSD","EOSUSD","TRXUSD","ADAUSD","BSVUSD","XLMUSD","GLDUSD","ZECUSD","XEMUSD"};
   double value=amount;
   for(int i=0;i<ArraySize(codes);i++)
   {
      // Independent checks, with a fresh account read for each comparison.
      if(AccountInfoString(ACCOUNT_CURRENCY)==codes[i] || AccountInfoString(ACCOUNT_CURRENCY)==lower[i])
      {
         if(Bot1CurrencyPair(amount,pairs[i]+suffix,i>=3 && i<=9,value)<0)
         {
            // The original stops before its caller can store this conversion.
            double zero=0.0;
            return amount/zero;
         }
      }
   }
   return value;
}
void Bot1LoadStrategyContext(const uint index,const Bot1StrategyContextInputs &inputs,
                             Bot1StrategyContextState &state)
{
   const string suffix[]={"_XAUUSD_1","_XAUUSD_2","_XAUUSD_3","_XAUUSD_4",
                          "_XAUUSD_5","_XAUUSD_6","_XAUUSD_7","_XAUUSD_8"};
   const double amounts[]={60.0,90.0,27.0,37.0,16.0,30.0,25.0,37.0};
   if(inputs.keep_comment==0) state.order_comment=inputs.comment_prefix+suffix[index];
   state.magic=(int)((uint)inputs.base_magic+index+(uint)1);
   state.converted_value=Bot1StrategyContextAmount(amounts[index],inputs.symbol_suffix);
}
#endif
