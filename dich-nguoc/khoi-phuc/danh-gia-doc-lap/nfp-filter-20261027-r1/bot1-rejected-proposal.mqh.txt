#ifndef BOT1_NFP_FILTER_MQH
#define BOT1_NFP_FILTER_MQH
#include "Bot1NfpCalendar.mqh"
#include "Bot1AmericanDST.mqh"
#include "Bot1NewsSelector.mqh"
#include "Bot1PendingDistanceCancel.mqh"
#include "Bot1PositionSelection.mqh"
struct Bot1NfpFilterSettings { uchar enabled,use_calendar,throttle,delete_pending,close_positions; int before_minutes,after_minutes; };
struct Bot1NfpFilterState { uchar initialization_notice,notice,blocked; long last_query,utc_reference; };
struct Bot1NfpExecution { int strategy_magic,magic_base; string symbol; double reference_a,reference_b,threshold_a,gap; ulong last_ticket,deviation; Bot1PendingRequestState trade; Bot1TradeSendOptions options; };
void Bot1NfpFilterActions(const Bot1NfpFilterSettings &s,Bot1NfpFilterState &st,Bot1NfpExecution &x){if(s.delete_pending!=0){Bot1CancelDistantPending(1,true,x.strategy_magic,x.symbol,x.reference_b,x.gap,x.last_ticket,x.trade,x.options);Bot1CancelDistantPending(1,false,x.strategy_magic,x.symbol,x.reference_a,x.gap,x.last_ticket,x.trade,x.options);}if(s.close_positions!=0){Bot1CloseFilteredBuyPositions(x.magic_base,x.symbol,x.last_ticket,x.trade,x.options,x.deviation,x.reference_a,x.reference_b,x.threshold_a,x.gap);Bot1CloseFilteredSellPositions(x.magic_base,x.symbol,x.last_ticket,x.trade,x.options,x.deviation,x.reference_a,x.reference_b,x.threshold_a,x.gap);}if(st.blocked==0)Print("NFP!! deleting trades!!");st.blocked=1;}
int Bot1NfpYear(){MqlDateTime t;TimeCurrent(t);return t.year;}int Bot1NfpMonth(){MqlDateTime t;TimeCurrent(t);return t.mon;}
void Bot1ProcessNfpFilter(const Bot1NfpFilterSettings &s,Bot1NfpFilterState &st,Bot1NewsEvents &e,const MqlDateTime &cal,const long &history[],Bot1NfpExecution &x){st.initialization_notice=0;if(s.enabled==0)return;if(s.use_calendar!=0&&MQLInfoInteger(MQL_TESTER)!=1){long now=TimeTradeServer();if(now>st.last_query+(s.throttle!=0?900:0)){e.nfp=Bot1FindNfpTime();st.last_query=TimeTradeServer();}if(e.nfp>0&&(st.notice&1)!=0){st.notice=0;Print("NFP filter: MQL5 Economic Calendar is available again, next release ",TimeToString((datetime)e.nfp,3));}long cur=TimeCurrent();if(e.nfp>0&&cur>=e.nfp+Bot1NewsMinutes32(-s.before_minutes)){long cur2=TimeCurrent();if(cur2<=e.nfp+Bot1NewsMinutes32(s.after_minutes))Bot1NfpFilterActions(s,st,x);else st.blocked=0;}}if(s.use_calendar!=0&&MQLInfoInteger(MQL_TESTER)!=1&&e.nfp!=0)return;if(s.use_calendar!=0&&MQLInfoInteger(MQL_TESTER)!=1&&st.notice==0){st.notice=1;Print("NFP filter: the MQL5 Economic Calendar returned no release - using the EA's built-in dates instead. Protection is still active. If this persists, open Toolbox -> Calendar in the terminal so it can sync.");}long offset=Bot1AmericanDST()?0:3600;int year=Bot1NfpYear();if(year<=2026){long selected=0;for(int i=0;i<300;i++){MqlDateTime d;TimeToStruct((datetime)history[i],d);if(d.year!=Bot1NfpYear())continue;TimeToStruct((datetime)history[i],d);if(d.mon!=Bot1NfpMonth())continue;selected=history[i];e.nfp=TimeTradeServer()+offset+selected-st.utc_reference;break;}long shifted=selected+offset;if(st.utc_reference<shifted+Bot1NewsMinutes32(-s.before_minutes)||st.utc_reference>shifted+Bot1NewsMinutes32(s.after_minutes))st.blocked=0;else Bot1NfpFilterActions(s,st,x);return;}if(cal.day>7||cal.day_of_week!=5)return;MqlDateTime d;TimeCurrent(d);string stamp=IntegerToString(year,0,' ')+IntegerToString(Bot1NfpMonth(),0,' ')+IntegerToString(d.day,0,' ')+" "+IntegerToString(1230,0,' ');long when=StringToTime(stamp);if(st.utc_reference<when+Bot1NewsMinutes32(-s.before_minutes)||st.utc_reference>when+Bot1NewsMinutes32(s.after_minutes))st.blocked=0;else Bot1NfpFilterActions(s,st,x);}
#endif
