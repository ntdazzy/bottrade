#ifndef BOT3_NFP_FILTER_MQH
#define BOT3_NFP_FILTER_MQH
#include "Bot3NfpCalendar.mqh"
#include "Bot3AmericanDST.mqh"
#include "Bot3NewsSelector.mqh"
#include "Bot3PendingDistanceCancel.mqh"
#include "Bot3PositionSelection.mqh"
struct Bot3NfpFilterSettings { uchar enabled,use_calendar,throttle,delete_pending,close_positions; int before_minutes,after_minutes; };
struct Bot3NfpFilterState { uchar initialization_notice,notice,blocked; long last_query,utc_reference; };
struct Bot3NfpExecution { int strategy_magic,magic_base; string symbol; double reference_a,reference_b,threshold_a,gap; ulong last_ticket,deviation; Bot3PendingRequestState trade; Bot3TradeSendOptions options; };
void Bot3NfpFilterActions(const Bot3NfpFilterSettings &s,Bot3NfpFilterState &st,Bot3NfpExecution &x){if(s.delete_pending!=0){Bot3CancelDistantPending(1,true,x.strategy_magic,x.symbol,x.reference_b,x.gap,x.last_ticket,x.trade,x.options);Bot3CancelDistantPending(1,false,x.strategy_magic,x.symbol,x.reference_a,x.gap,x.last_ticket,x.trade,x.options);}if(s.close_positions!=0){Bot3CloseFilteredBuyPositions(x.magic_base,x.symbol,x.last_ticket,x.trade,x.options,x.deviation,x.reference_a,x.reference_b,x.threshold_a,x.gap);Bot3CloseFilteredSellPositions(x.magic_base,x.symbol,x.last_ticket,x.trade,x.options,x.deviation,x.reference_a,x.reference_b,x.threshold_a,x.gap);}if(st.blocked==0)Print("NFP!! deleting trades!!");st.blocked=1;}
int Bot3NfpYear(){MqlDateTime t;TimeCurrent(t);return t.year;}int Bot3NfpMonth(){MqlDateTime t;TimeCurrent(t);return t.mon;}
void Bot3ProcessNfpFilter(const Bot3NfpFilterSettings &s,Bot3NfpFilterState &st,Bot3NewsEvents &e,const MqlDateTime &cal,const long &history[],Bot3NfpExecution &x){st.initialization_notice=0;if(s.enabled==0)return;if(s.use_calendar!=0&&MQLInfoInteger(MQL_TESTER)!=1){long now=TimeTradeServer();if(now>st.last_query+(s.throttle!=0?900:0)){e.nfp=Bot3FindNfpTime();st.last_query=TimeTradeServer();}if(e.nfp>0&&(st.notice&1)!=0){st.notice=0;Print("NFP filter: MQL5 Economic Calendar is available again, next release ",TimeToString((datetime)e.nfp,3));}long cur=TimeCurrent();if(e.nfp>0&&cur>=e.nfp+Bot3NewsMinutes32(-s.before_minutes)){long cur2=TimeCurrent();if(cur2<=e.nfp+Bot3NewsMinutes32(s.after_minutes))Bot3NfpFilterActions(s,st,x);else st.blocked=0;}}if(s.use_calendar!=0&&MQLInfoInteger(MQL_TESTER)!=1&&e.nfp!=0)return;if(s.use_calendar!=0&&MQLInfoInteger(MQL_TESTER)!=1&&st.notice==0){st.notice=1;Print("NFP filter: the MQL5 Economic Calendar returned no release - using the EA's built-in dates instead. Protection is still active. If this persists, open Toolbox -> Calendar in the terminal so it can sync.");}long offset=Bot3AmericanDST()?0:3600;int year=Bot3NfpYear();if(year<=2026){long selected=0;for(int i=0;i<300;i++){MqlDateTime d;TimeToStruct((datetime)history[i],d);if(d.year!=Bot3NfpYear())continue;TimeToStruct((datetime)history[i],d);if(d.mon!=Bot3NfpMonth())continue;selected=history[i];e.nfp=TimeTradeServer()+offset+selected-st.utc_reference;break;}long shifted=selected+offset;if(st.utc_reference<shifted+Bot3NewsMinutes32(-s.before_minutes)||st.utc_reference>shifted+Bot3NewsMinutes32(s.after_minutes))st.blocked=0;else Bot3NfpFilterActions(s,st,x);return;}if(cal.day>7||cal.day_of_week!=5)return;MqlDateTime d;TimeCurrent(d);string stamp=IntegerToString(year,0,' ')+IntegerToString(Bot3NfpMonth(),0,' ')+IntegerToString(d.day,0,' ')+" "+IntegerToString(1230,0,' ');long when=StringToTime(stamp);if(st.utc_reference<when+Bot3NewsMinutes32(-s.before_minutes)||st.utc_reference>when+Bot3NewsMinutes32(s.after_minutes))st.blocked=0;else Bot3NfpFilterActions(s,st,x);}
#endif
